    48→/**
    49→ * Dynamic update badge — shows the freshest date across ALL data sources,
    50→ * with a live indicator and lag label. Falls back to headline date if
    51→ * metadata hasn't loaded yet.
    52→ */
    53→function DynamicUpdateBadge({
    54→  meta,
    55→  headlineLatest,
    56→}: {
    57→  meta: DataMetadata | null;
    58→  headlineLatest?: string;
    59→}) {
    60→  // Determine what to show
    61→  const freshestDate = meta?.freshest_date || headlineLatest || '';
    62→  const freshestSource = meta?.freshest_source || '';
    63→
    64→  if (!freshestDate) return null;
    65→
    66→  // Compute lag
    67→  let lagLabel = '';
    68→  let lagColor = 'text-[var(--text-faint)]';
    69→  try {
    70→    const dateMs = new Date(freshestDate + 'T00:00:00').getTime();
    71→    const nowMs = Date.now();
    72→    const hoursAgo = differenceInHours(nowMs, dateMs);
    73→    if (hoursAgo < 24) {
    74→      lagLabel = `${hoursAgo}h ago`;
    75→      lagColor = 'text-[#85AB8B]';
    76→    } else if (hoursAgo < 48) {
    77→      lagLabel = `${Math.round(hoursAgo / 24)}d ago`;
    78→      lagColor = 'text-[#85AB8B]';
    79→    } else {
    80→      const days = Math.round(hoursAgo / 24);
    81→      lagLabel = `${days}d ago`;
    82→      lagColor = days > 15 ? 'text-orange-400' : 'text-amber-400';
    83→    }
    84→  } catch {
    85→    lagLabel = freshestDate;
    86→  }
    87→
    88→  return (
    89→    <p className="mt-2 text-[10px] sm:text-[11px] text-[var(--text-ghost)] flex items-center justify-center gap-2 flex-wrap">
    90→      <span className="flex items-center gap-1.5">
    91→        <span className="relative flex h-2 w-2">
    92→          <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-[#85AB8B] opacity-60" />
    93→          <span className="relative inline-flex rounded-full h-2 w-2 bg-[#85AB8B]" />
    94→        </span>
    95→        <span className="text-[var(--text-faint)]">
    96→          Data via <span className="text-[var(--text-muted)]">data.gov.my</span>
    97→        </span>
    98→      </span>
    99→      <span className="text-[var(--text-ghost)]">·</span>
   100→      <span className="flex items-center gap-1.5">
   101→        <RefreshCw className="w-3 h-3 text-[var(--text-ghost)]" />
   102→        <span className={`font-medium tabular-nums ${lagColor}`}>
   103→          {freshestDate}
   104→        </span>
   105→        {freshestSource && (
   106→          <span className="text-[var(--text-ghost)]">
   107→            ({freshestSource} · {lagLabel})
   108→          </span>
   109→        )}
   110→      </span>
   111→    </p>
   112→  );
   113→}
