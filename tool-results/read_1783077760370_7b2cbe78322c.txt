    55→/**
    56→ * Dynamic update badge — shows the freshest date across ALL data sources,
    57→ * with a live indicator and lag label. Falls back to headline date if
    58→ * metadata hasn't loaded yet.
    59→ */
    60→function DynamicUpdateBadge({
    61→  meta,
    62→  headlineLatest,
    63→}: {
    64→  meta: DataMetadata | null;
    65→  headlineLatest?: string;
    66→}) {
    67→  // Determine what to show
    68→  const freshestDate = meta?.freshest_date || headlineLatest || '';
    69→  const freshestSource = meta?.freshest_source || '';
    70→
    71→  if (!freshestDate) return null;
    72→
    73→  // Compute lag
    74→  let lagLabel = '';
    75→  let lagColor = 'text-[var(--text-faint)]';
    76→  try {
    77→    const dateMs = new Date(freshestDate + 'T00:00:00').getTime();
    78→    const nowMs = Date.now();
    79→    const hoursAgo = differenceInHours(nowMs, dateMs);
    80→    if (hoursAgo < 24) {
    81→      lagLabel = `${hoursAgo}h ago`;
    82→      lagColor = 'text-[#85AB8B]';
    83→    } else if (hoursAgo < 48) {
    84→      lagLabel = `${Math.round(hoursAgo / 24)}d ago`;
    85→      lagColor = 'text-[#85AB8B]';
    86→    } else {
    87→      const days = Math.round(hoursAgo / 24);
    88→      lagLabel = `${days}d ago`;
    89→      lagColor = days > 15 ? 'text-orange-400' : 'text-amber-400';
    90→    }
    91→  } catch {
    92→    lagLabel = freshestDate;
    93→  }
    94→
    95→  return (
    96→    <p className="mt-2 text-[10px] sm:text-[11px] text-[var(--text-ghost)] flex items-center justify-center gap-2 flex-wrap">
    97→      <span className="flex items-center gap-1.5">
    98→        <span className="relative flex h-2 w-2">
    99→          <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-[#85AB8B] opacity-60" />
   100→          <span className="relative inline-flex rounded-full h-2 w-2 bg-[#85AB8B]" />
   101→        </span>
   102→        <span className="text-[var(--text-faint)]">
   103→          Data via <span className="text-[var(--text-muted)]">data.gov.my</span>
   104→        </span>
   105→      </span>
   106→      <span className="text-[var(--text-ghost)]">·</span>
   107→      <span className="flex items-center gap-1.5">
   108→        <RefreshCw className="w-3 h-3 text-[var(--text-ghost)]" />
   109→        <span className={`font-medium tabular-nums ${lagColor}`}>
   110→          {freshestDate}
   111→        </span>
   112→        {freshestSource && (
   113→          <span className="text-[var(--text-ghost)]">
   114→            ({freshestSource} · {lagLabel})
   115→          </span>
   116→        )}
   117→      </span>
   118→    </p>
   119→  );
   120→}

   925→// Lightweight pub/sub for landing state (avoids storage event cross-tab limitation)
   926→const landingListeners: Array<() => void> = [];
   927→function subscribeLanding(cb: () => void) {
   928→  landingListeners.push(cb);
   929→  return () => {
   930→    const idx = landingListeners.indexOf(cb);
   931→    if (idx > -1) landingListeners.splice(idx, 1);
   932→  };
   933→}
   934→function getLandingSnapshot() {
   935→  return localStorage.getItem('rapidstats-landing-dismissed') === '1';
   936→}
   937→function getLandingServerSnapshot() {
   938→  return false;
   939→}
   940→
   941→export default function Home() {
   942→  const landingDismissed = useSyncExternalStore(
   943→    subscribeLanding,
   944→    getLandingSnapshot,
   945→    getLandingServerSnapshot
   946→  );
   947→
   948→  const handleEnter = useCallback(() => {
   949→    localStorage.setItem('rapidstats-landing-dismissed', '1');
   950→    landingListeners.forEach((l) => l());
   951→  }, []);
   952→
   953→  if (!landingDismissed) {
   954→    return <LandingPage onEnter={handleEnter} />;
   955→  }
   956→
   957→  return <DashboardView />;
   958→}
