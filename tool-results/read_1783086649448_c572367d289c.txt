    57→/**
    58→ * Dynamic update badge — shows the freshest date across ALL data sources,
    59→ * with a live indicator and lag label. Falls back to headline date if
    60→ * metadata hasn't loaded yet.
    61→ */
    62→function DynamicUpdateBadge({
    63→  meta,
    64→  headlineLatest,
    65→}: {
    66→  meta: DataMetadata | null;
    67→  headlineLatest?: string;
    68→}) {
    69→  // Determine what to show
    70→  const freshestDate = meta?.freshest_date || headlineLatest || '';
    71→  const freshestSource = meta?.freshest_source || '';
    72→
    73→  if (!freshestDate) return null;
    74→
    75→  // Compute lag
    76→  let lagLabel = '';
    77→  let lagColor = 'text-[var(--text-faint)]';
    78→  try {
    79→    const dateMs = new Date(freshestDate + 'T00:00:00').getTime();
    80→    const nowMs = Date.now();
    81→    const hoursAgo = differenceInHours(nowMs, dateMs);
    82→    if (hoursAgo < 24) {
    83→      lagLabel = `${hoursAgo}h ago`;
    84→      lagColor = 'text-[#85AB8B]';
    85→    } else if (hoursAgo < 48) {
    86→      lagLabel = `${Math.round(hoursAgo / 24)}d ago`;
    87→      lagColor = 'text-[#85AB8B]';
    88→    } else {
    89→      const days = Math.round(hoursAgo / 24);
    90→      lagLabel = `${days}d ago`;
    91→      lagColor = days > 15 ? 'text-orange-400' : 'text-amber-400';
    92→    }
    93→  } catch {
    94→    lagLabel = freshestDate;
    95→  }
    96→
    97→  return (
    98→    <p className="mt-2 text-[10px] sm:text-[11px] text-[var(--text-ghost)] flex items-center justify-center gap-2 flex-wrap">
    99→      <span className="flex items-center gap-1.5">
   100→        <span className="relative flex h-2 w-2">
   101→          <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-[#85AB8B] opacity-60" />
   102→          <span className="relative inline-flex rounded-full h-2 w-2 bg-[#85AB8B]" />
   103→        </span>
   104→        <span className="text-[var(--text-faint)]">
   105→          Data via <span className="text-[var(--text-muted)]">data.gov.my</span>
   106→        </span>
   107→      </span>
   108→      <span className="text-[var(--text-ghost)]">·</span>
   109→      <span className="flex items-center gap-1.5">
   110→        <RefreshCw className="w-3 h-3 text-[var(--text-ghost)]" />
   111→        <span className={`font-medium tabular-nums ${lagColor}`}>
   112→          {freshestDate}
   113→        </span>
   114→        {freshestSource && (
   115→          <span className="text-[var(--text-ghost)]">
   116→            ({freshestSource} · {lagLabel})
   117→          </span>
   118→        )}
   119→      </span>
   120→    </p>
   121→  );
   122→}

   937→// Lightweight pub/sub for landing state (avoids storage event cross-tab limitation)
   938→const landingListeners: Array<() => void> = [];
   939→function subscribeLanding(cb: () => void) {
   940→  landingListeners.push(cb);
   941→  return () => {
   942→    const idx = landingListeners.indexOf(cb);
   943→    if (idx > -1) landingListeners.splice(idx, 1);
   944→  };
   945→}
   946→function getLandingSnapshot() {
   947→  return localStorage.getItem('rapidstats-landing-dismissed') === '1';
   948→}
   949→function getLandingServerSnapshot() {
   950→  return false;
   951→}
   952→
   953→export default function Home() {
   954→  const landingDismissed = useSyncExternalStore(
   955→    subscribeLanding,
   956→    getLandingSnapshot,
   957→    getLandingServerSnapshot
   958→  );
   959→
   960→  const handleEnter = useCallback(() => {
   961→    localStorage.setItem('rapidstats-landing-dismissed', '1');
   962→    landingListeners.forEach((l) => l());
   963→  }, []);
   964→
   965→  if (!landingDismissed) {
   966→    return <LandingPage onEnter={handleEnter} />;
   967→  }
   968→
   969→  return <DashboardView />;
   970→}
