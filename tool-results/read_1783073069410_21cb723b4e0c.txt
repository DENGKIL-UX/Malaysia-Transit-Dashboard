    54→/**
    55→ * Dynamic update badge — shows the freshest date across ALL data sources,
    56→ * with a live indicator and lag label. Falls back to headline date if
    57→ * metadata hasn't loaded yet.
    58→ */
    59→function DynamicUpdateBadge({
    60→  meta,
    61→  headlineLatest,
    62→}: {
    63→  meta: DataMetadata | null;
    64→  headlineLatest?: string;
    65→}) {
    66→  // Determine what to show
    67→  const freshestDate = meta?.freshest_date || headlineLatest || '';
    68→  const freshestSource = meta?.freshest_source || '';
    69→
    70→  if (!freshestDate) return null;
    71→
    72→  // Compute lag
    73→  let lagLabel = '';
    74→  let lagColor = 'text-[var(--text-faint)]';
    75→  try {
    76→    const dateMs = new Date(freshestDate + 'T00:00:00').getTime();
    77→    const nowMs = Date.now();
    78→    const hoursAgo = differenceInHours(nowMs, dateMs);
    79→    if (hoursAgo < 24) {
    80→      lagLabel = `${hoursAgo}h ago`;
    81→      lagColor = 'text-[#85AB8B]';
    82→    } else if (hoursAgo < 48) {
    83→      lagLabel = `${Math.round(hoursAgo / 24)}d ago`;
    84→      lagColor = 'text-[#85AB8B]';
    85→    } else {
    86→      const days = Math.round(hoursAgo / 24);
    87→      lagLabel = `${days}d ago`;
    88→      lagColor = days > 15 ? 'text-orange-400' : 'text-amber-400';
    89→    }
    90→  } catch {
    91→    lagLabel = freshestDate;
    92→  }
    93→
    94→  return (
    95→    <p className="mt-2 text-[10px] sm:text-[11px] text-[var(--text-ghost)] flex items-center justify-center gap-2 flex-wrap">
    96→      <span className="flex items-center gap-1.5">
    97→        <span className="relative flex h-2 w-2">
    98→          <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-[#85AB8B] opacity-60" />
    99→          <span className="relative inline-flex rounded-full h-2 w-2 bg-[#85AB8B]" />
   100→        </span>
   101→        <span className="text-[var(--text-faint)]">
   102→          Data via <span className="text-[var(--text-muted)]">data.gov.my</span>
   103→        </span>
   104→      </span>
   105→      <span className="text-[var(--text-ghost)]">·</span>
   106→      <span className="flex items-center gap-1.5">
   107→        <RefreshCw className="w-3 h-3 text-[var(--text-ghost)]" />
   108→        <span className={`font-medium tabular-nums ${lagColor}`}>
   109→          {freshestDate}
   110→        </span>
   111→        {freshestSource && (
   112→          <span className="text-[var(--text-ghost)]">
   113→            ({freshestSource} · {lagLabel})
   114→          </span>
   115→        )}
   116→      </span>
   117→    </p>
   118→  );
   119→}
