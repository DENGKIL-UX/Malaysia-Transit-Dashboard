    51→/**
    52→ * Dynamic update badge — shows the freshest date across ALL data sources,
    53→ * with a live indicator and lag label. Falls back to headline date if
    54→ * metadata hasn't loaded yet.
    55→ */
    56→function DynamicUpdateBadge({
    57→  meta,
    58→  headlineLatest,
    59→}: {
    60→  meta: DataMetadata | null;
    61→  headlineLatest?: string;
    62→}) {
    63→  // Determine what to show
    64→  const freshestDate = meta?.freshest_date || headlineLatest || '';
    65→  const freshestSource = meta?.freshest_source || '';
    66→
    67→  if (!freshestDate) return null;
    68→
    69→  // Compute lag
    70→  let lagLabel = '';
    71→  let lagColor = 'text-[var(--text-faint)]';
    72→  try {
    73→    const dateMs = new Date(freshestDate + 'T00:00:00').getTime();
    74→    const nowMs = Date.now();
    75→    const hoursAgo = differenceInHours(nowMs, dateMs);
    76→    if (hoursAgo < 24) {
    77→      lagLabel = `${hoursAgo}h ago`;
    78→      lagColor = 'text-[#85AB8B]';
    79→    } else if (hoursAgo < 48) {
    80→      lagLabel = `${Math.round(hoursAgo / 24)}d ago`;
    81→      lagColor = 'text-[#85AB8B]';
    82→    } else {
    83→      const days = Math.round(hoursAgo / 24);
    84→      lagLabel = `${days}d ago`;
    85→      lagColor = days > 15 ? 'text-orange-400' : 'text-amber-400';
    86→    }
    87→  } catch {
    88→    lagLabel = freshestDate;
    89→  }
    90→
    91→  return (
    92→    <p className="mt-2 text-[10px] sm:text-[11px] text-[var(--text-ghost)] flex items-center justify-center gap-2 flex-wrap">
    93→      <span className="flex items-center gap-1.5">
    94→        <span className="relative flex h-2 w-2">
    95→          <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-[#85AB8B] opacity-60" />
    96→          <span className="relative inline-flex rounded-full h-2 w-2 bg-[#85AB8B]" />
    97→        </span>
    98→        <span className="text-[var(--text-faint)]">
    99→          Data via <span className="text-[var(--text-muted)]">data.gov.my</span>
   100→        </span>
   101→      </span>
   102→      <span className="text-[var(--text-ghost)]">·</span>
   103→      <span className="flex items-center gap-1.5">
   104→        <RefreshCw className="w-3 h-3 text-[var(--text-ghost)]" />
   105→        <span className={`font-medium tabular-nums ${lagColor}`}>
   106→          {freshestDate}
   107→        </span>
   108→        {freshestSource && (
   109→          <span className="text-[var(--text-ghost)]">
   110→            ({freshestSource} · {lagLabel})
   111→          </span>
   112→        )}
   113→      </span>
   114→    </p>
   115→  );
   116→}
