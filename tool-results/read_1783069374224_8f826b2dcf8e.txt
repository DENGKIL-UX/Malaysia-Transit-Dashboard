    51→    51→/**
    52→    52→ * Dynamic update badge — shows the freshest date across ALL data sources,
    53→    53→ * with a live indicator and lag label. Falls back to headline date if
    54→    54→ * metadata hasn't loaded yet.
    55→    55→ */
    56→    56→function DynamicUpdateBadge({
    57→    57→  meta,
    58→    58→  headlineLatest,
    59→    59→}: {
    60→    60→  meta: DataMetadata | null;
    61→    61→  headlineLatest?: string;
    62→    62→}) {
    63→    63→  // Determine what to show
    64→    64→  const freshestDate = meta?.freshest_date || headlineLatest || '';
    65→    65→  const freshestSource = meta?.freshest_source || '';
    66→    66→
    67→    67→  if (!freshestDate) return null;
    68→    68→
    69→    69→  // Compute lag
    70→    70→  let lagLabel = '';
    71→    71→  let lagColor = 'text-[var(--text-faint)]';
    72→    72→  try {
    73→    73→    const dateMs = new Date(freshestDate + 'T00:00:00').getTime();
    74→    74→    const nowMs = Date.now();
    75→    75→    const hoursAgo = differenceInHours(nowMs, dateMs);
    76→    76→    if (hoursAgo < 24) {
    77→    77→      lagLabel = `${hoursAgo}h ago`;
    78→    78→      lagColor = 'text-[#85AB8B]';
    79→    79→    } else if (hoursAgo < 48) {
    80→    80→      lagLabel = `${Math.round(hoursAgo / 24)}d ago`;
    81→    81→      lagColor = 'text-[#85AB8B]';
    82→    82→    } else {
    83→    83→      const days = Math.round(hoursAgo / 24);
    84→    84→      lagLabel = `${days}d ago`;
    85→    85→      lagColor = days > 15 ? 'text-orange-400' : 'text-amber-400';
    86→    86→    }
    87→    87→  } catch {
    88→    88→    lagLabel = freshestDate;
    89→    89→  }
    90→    90→
    91→    91→  return (
    92→    92→    <p className="mt-2 text-[10px] sm:text-[11px] text-[var(--text-ghost)] flex items-center justify-center gap-2 flex-wrap">
    93→    93→      <span className="flex items-center gap-1.5">
    94→    94→        <span className="relative flex h-2 w-2">
    95→    95→          <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-[#85AB8B] opacity-60" />
    96→    96→          <span className="relative inline-flex rounded-full h-2 w-2 bg-[#85AB8B]" />
    97→    97→        </span>
    98→    98→        <span className="text-[var(--text-faint)]">
    99→    99→          Data via <span className="text-[var(--text-muted)]">data.gov.my</span>
   100→   100→        </span>
   101→   101→      </span>
   102→   102→      <span className="text-[var(--text-ghost)]">·</span>
   103→   103→      <span className="flex items-center gap-1.5">
   104→   104→        <RefreshCw className="w-3 h-3 text-[var(--text-ghost)]" />
   105→   105→        <span className={`font-medium tabular-nums ${lagColor}`}>
   106→   106→          {freshestDate}
   107→   107→        </span>
   108→   108→        {freshestSource && (
   109→   109→          <span className="text-[var(--text-ghost)]">
   110→   110→            ({freshestSource} · {lagLabel})
   111→   111→          </span>
   112→   112→        )}
   113→   113→      </span>
   114→   114→    </p>
   115→   115→  );
   116→   116→}
