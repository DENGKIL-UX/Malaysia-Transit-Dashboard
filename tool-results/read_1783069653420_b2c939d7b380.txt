   127→// ─── Window Computation ──────────────────────────────────────────────
   128→
   129→function computeWindows<T extends { date: string }>(data: T[]): WindowDef[] {
   130→  if (!data.length) return [];
   131→
   132→  // Extract and sort unique dates
   133→  const dates = data.map((d) => d.date.split(' ')[0]).sort();
   134→  const latestStr = dates[dates.length - 1];
   135→  const earliestStr = dates[0];
   136→
   137→  const latest = new Date(latestStr + 'T00:00:00');
   138→  const earliest = new Date(earliestStr + 'T00:00:00');
   139→
   140→  const windows: WindowDef[] = [];
   141→  let windowEnd = new Date(latest);
   142→
   143→  while (true) {
   144→    const windowStart = subDays(windowEnd, WINDOW_SIZE - 1);
   145→    const startMs = windowStart.getTime();
   146→    const endMs = windowEnd.getTime();
   147→
   148→    if (startMs < earliest.getTime()) {
   149→      // Last (partial) window — include only if it meets minimum threshold
   150→      const actualDays = Math.round((endMs - earliest.getTime()) / 864e5) + 1;
   151→      if (actualDays >= MIN_WINDOW_DAYS) {
   152→        windows.push({
   153→          start: earliestStr,
   154→          end: format(windowEnd, 'yyyy-MM-dd'),
   155→          days: actualDays,
   156→        });
   157→      }
   158→      break;
   159→    }
   160→
   161→    windows.push({
   162→      start: format(windowStart, 'yyyy-MM-dd'),
   163→      end: format(windowEnd, 'yyyy-MM-dd'),
   164→      days: WINDOW_SIZE,
   165→    });
   166→
   167→    // Next window ends the day before this one starts
   168→    windowEnd = subDays(windowStart, 1);
   169→  }
   170→
   171→  return windows; // index 0 = latest
   172→}

   174→// ─── Day-Type Computation ───────────────────────────────────────────
   175→
   176→function computeDayTypeAverages<T extends { date: string }>(
   177→  data: T[],
   178→  keys: string[],
   179→  windowStart?: string,
   180→  windowEnd?: string
   181→): DayTypeResult {
   182→  if (!data.length) {
   183→    return {
   184→      chartData: [],
   185→      weekdayAvg: 0,
   186→      peakDay: 'N/A',
   187→      peakValue: 0,
   188→      lowDay: 'N/A',
   189→      lowValue: 0,
   190→      weekendWeekdayRatio: 0,
   191→      windowDays: 0,
   192→      windowStart: '',
   193→      windowEnd: '',
   194→      perServiceWeekdayAvg: {},
   195→    };
   196→  }
   197→
   198→  // Filter to explicit window
   199→  const filtered = data.filter((d) => {
   200→    const dateStr = d.date.split(' ')[0];
   201→    if (windowStart && dateStr < windowStart) return false;
   202→    if (windowEnd && dateStr > windowEnd) return false;
   203→    return true;
   204→  });
   205→
   206→  // ── Daily-total-first approach ──
   207→  // For each day, compute the daily total first, then group by DOW.
   208→  // This is robust to partial data (missing services on some days),
   209→  // whereas averaging per-service-then-summing drifts with incomplete records.
   210→  //
   211→  // DOW convention: Mon=0 … Sun=6 (chart axis order).
   212→  // NOTE: JS getDay() returns Sun=0 … Sat=6; we remap below.
   213→
   214→  interface DayRow {
   215→    dow: number; // Mon=0, Sun=6
   216→    dailyTotal: number;
   217→    perService: Record<string, number>;
   218→  }
   219→
   220→  const dayRows: DayRow[] = [];
   221→  for (const row of filtered) {
   222→    const dateStr = row.date.split(' ')[0];
   223→    const dt = new Date(dateStr + 'T00:00:00');
   224→    let dow = dt.getDay() - 1; // JS getDay(): Sun=0 → Mon=-1, so remap
   225→    if (dow < 0) dow = 6; // Sunday → index 6
   226→
   227→    const perService: Record<string, number> = {};
   228→    let dailyTotal = 0;
   229→    for (const key of keys) {
   230→      const val = (row as unknown as Record<string, unknown>)[key];
   231→      if (typeof val === 'number') {
   232→        perService[key] = val;
   233→        dailyTotal += val;
   234→      }
   235→    }
   236→    dayRows.push({ dow, dailyTotal, perService });
   237→  }
   238→
   239→  // Group daily totals by DOW for accurate statistics
   240→  const dowTotals: number[][] = Array.from({ length: 7 }, () => []);
   241→
   242→  // Group per-service values by DOW for stacked bar averages
   243→  const dowServiceBuckets: Record<number, Record<string, number[]>> = {};
   244→  for (let i = 0; i < 7; i++) {
   245→    dowServiceBuckets[i] = Object.fromEntries(keys.map((k) => [k, [] as number[]]));
   246→  }
   247→
   248→  for (const dr of dayRows) {
   249→    dowTotals[dr.dow].push(dr.dailyTotal);
   250→    for (const key of keys) {
   251→      if (dr.perService[key] !== undefined) {
   252→        dowServiceBuckets[dr.dow][key].push(dr.perService[key]);
   253→      }
   254→    }
   255→  }
   256→
   257→  // Compute chart data: per-service averages for stacked bars,
   258→  // daily-total average for the `total` field used in stats.
   259→  const chartData = DAY_NAMES.map((day, i) => {
   260→    const entry: Record<string, number | string> = {
   261→      day,
   262→      dayLabel: DAY_FULL[i],
   263→    };
   264→
   265→    // Per-service averages (for stacked bar visualization)
   266→    for (const key of keys) {
   267→      const vals = dowServiceBuckets[i][key];
   268→      entry[key] =
   269→        vals.length > 0 ? Math.round(vals.reduce((s, v) => s + v, 0) / vals.length) : 0;
   270→    }
   271→
   272→    // Daily-total-first average (statistically correct baseline)
   273→    const dailyTotals = dowTotals[i];
   274→    entry.total =
   275→      dailyTotals.length > 0
   276→        ? Math.round(dailyTotals.reduce((s, v) => s + v, 0) / dailyTotals.length)
   277→        : 0;
   278→
   279→    return entry;
   280→  });
   281→
   282→  // Weekday average (Mon-Fri, indices 0-4) — from daily totals
   283→  const weekdayDailyTotals = dayRows
   284→    .filter((dr) => dr.dow >= 0 && dr.dow <= 4)
   285→    .map((dr) => dr.dailyTotal);
   286→  const weekdayAvg =
   287→    weekdayDailyTotals.length > 0
   288→      ? Math.round(
   289→          weekdayDailyTotals.reduce((s, v) => s + v, 0) / weekdayDailyTotals.length
   290→        )
   291→      : 0;
   292→
   293→  // Per-service weekday averages (for tooltip breakdown)
   294→  const perServiceWeekdayAvg: Record<string, number> = {};
   295→  for (const key of keys) {
   296→    const allVals: number[] = [];
   297→    for (let i = 0; i <= 4; i++) {
   298→      allVals.push(...dowServiceBuckets[i][key]);
   299→    }
   300→    perServiceWeekdayAvg[key] =
   301→      allVals.length > 0 ? Math.round(allVals.reduce((s, v) => s + v, 0) / allVals.length) : 0;
   302→  }
   303→
   304→  // Peak / low / weekend ratio — all from daily-total averages
   305→  const totals = chartData.map((d) => d.total as number);
   306→  const peakIdx = totals.indexOf(Math.max(...totals));
   307→  const lowIdx = totals.indexOf(Math.min(...totals));
   308→
   309→  const weekendDailyTotals = dayRows
   310→    .filter((dr) => dr.dow === 5 || dr.dow === 6)
   311→    .map((dr) => dr.dailyTotal);
   312→  const weekendAvg =
   313→    weekendDailyTotals.length > 0
   314→      ? weekendDailyTotals.reduce((s, v) => s + v, 0) / weekendDailyTotals.length
   315→      : 0;
   316→  const weekendWeekdayRatio =
   317→    weekdayAvg > 0 ? Math.round((weekendAvg / weekdayAvg) * 100) / 100 : 0;
   318→
   319→  return {
   320→    chartData,
   321→    weekdayAvg,
   322→    peakDay: DAY_FULL[peakIdx],
   323→    peakValue: totals[peakIdx],
   324→    lowDay: DAY_FULL[lowIdx],
   325→    lowValue: totals[lowIdx],
   326→    weekendWeekdayRatio,
   327→    windowDays: filtered.length,
   328→    windowStart: filtered[0]?.date.split(' ')[0] ?? '',
   329→    windowEnd: filtered[filtered.length - 1]?.date.split(' ')[0] ?? '',
   330→    perServiceWeekdayAvg,
   331→  };
   332→}

   334→// ─── Tooltip ─────────────────────────────────────────────────────────
   335→
   336→interface TooltipPayloadItem {
   337→  name: string;
   338→  value: number;
   339→  color: string;
   340→  payload?: {
   341→    dayLabel?: string;
   342→    total?: number;
   343→  };
   344→}
   345→
   346→function DayTypeTooltip({
   347→  active,
   348→  payload,
   349→  weekdayAvg,
   350→}: {
   351→  active?: boolean;
   352→  payload?: TooltipPayloadItem[];
   353→  weekdayAvg: number;
   354→}) {
   355→  if (!active || !payload?.length) return null;
   356→  const pl = payload[0]?.payload;
   357→  const total = payload.reduce((sum, p) => sum + p.value, 0);
   358→  const totalVsAvg =
   359→    weekdayAvg > 0 ? Math.round(((total - weekdayAvg) / weekdayAvg) * 100) / 100 : 0;
   360→
   361→  return (
   362→    <div className="bg-[var(--bg-tooltip)] backdrop-blur-md border border-[var(--border-subtle)] rounded-xl p-3 shadow-xl min-w-[240px]">
   363→      <p className="text-[10px] font-medium text-[#85AB8B] uppercase tracking-widest mb-1">
   364→        {pl?.dayLabel ?? ''}
   365→      </p>
   366→      <div className="space-y-1">
   367→        {payload.map((item) => (
   368→          <div key={item.name} className="flex items-center justify-between gap-6">
   369→            <div className="flex items-center gap-2">
   370→              <span
   371→                className="w-2 h-2 rounded-full shrink-0"
   372→                style={{ backgroundColor: item.color }}
   373→              />
   374→              <span className="text-[11px] text-[var(--text-muted)]">
   375→                {item.name}
   376→              </span>
   377→            </div>
   378→            <span className="text-[11px] font-semibold text-[var(--text-primary)] tabular-nums">
   379→              {item.value.toLocaleString()}
   380→            </span>
   381→          </div>
   382→        ))}
   383→        <div className="border-t border-[var(--border-faint)] pt-1 mt-1">
   384→          <div className="flex items-center justify-between gap-6">
   385→            <span className="text-[11px] font-medium text-[var(--text-secondary)]">
   386→              Total
   387→            </span>
   388→            <span className="text-[11px] font-bold text-[var(--text-primary)] tabular-nums">
   389→              {total.toLocaleString()}
   390→            </span>
   391→          </div>
   392→          <div className="flex items-center justify-between gap-6 mt-0.5">
   393→            <span className="text-[10px] text-[var(--text-faint)]">vs Wkday Avg</span>
   394→            <span
   395→              className={`text-[10px] font-semibold tabular-nums ${
   396→                totalVsAvg >= 0 ? 'text-emerald-400' : 'text-red-400'
   397→              }`}
   398→            >
   399→              {totalVsAvg >= 0 ? '+' : ''}
   400→              {(totalVsAvg * 100).toFixed(1)}%
   401→            </span>
   402→          </div>
   403→        </div>
   404→      </div>
   405→    </div>
   406→  );
   407→}

   409→// ─── Main Component ──────────────────────────────────────────────────

   411→export function DayTypeAnalytics() {
   412→  const [ktmbData, setKtmbData] = useState<KtmbRow[]>([]);
   413→  const [prasData, setPrasData] = useState<PrasaranaRow[]>([]);
   414→  const [activeTab, setActiveTab] = useState<'ktmb' | 'prasarana'>('ktmb');
   415→  const [hiddenLines, setHiddenLines] = useState<Set<string>>(new Set());
   416→  const [loading, setLoading] = useState(true);
   417→  const [windowOffset, setWindowOffset] = useState(0); // 0 = latest window
   418→
   419→  // Centralized metadata from Zustand
   420→  const meta = useAppStore((s) => s.metadata);
   421→
   422→  // Fetch data on mount
   423→  const fetchData = useCallback(async () => {
   424→    setLoading(true);
   425→    try {
   426→      const [ktmb, pras] = await Promise.all([
   427→        fetch('/ktmb-daily.json').then((r) => r.json()),
   428→        fetch('/prasarana-daily.json').then((r) => r.json()),
   429→      ]);
   430→      setKtmbData(ktmb);
   431→      setPrasData(pras);
   432→    } catch {
   433→      // Silently fail
   434→    } finally {
   435→      setLoading(false);
   436→    }
   437→  }, []);
   438→
   439→  useEffect(() => {
   440→    fetchData();
   441→  }, [fetchData]);
   442→
   443→  // Compute available windows for each data source
   444→  const ktmbWindows = useMemo(() => computeWindows(ktmbData), [ktmbData]);
   445→  const prasWindows = useMemo(() => computeWindows(prasData), [prasData]);
   446→
   447→  // Active windows based on tab
   448→  const windows = activeTab === 'ktmb' ? ktmbWindows : prasWindows;
   449→
   450→  // Clamp offset to available windows
   451→  const safeOffset = Math.min(windowOffset, Math.max(0, windows.length - 1));
   452→  const activeWindow = windows[safeOffset] ?? windows[0];
   453→
   454→  // Reset offset when tab changes
   455→  useEffect(() => {
   456→    setWindowOffset(0);
   457→  }, [activeTab]);
   458→
   459→  // Navigation
   460→  const canGoPrev = safeOffset < windows.length - 1;
   461→  const canGoNext = safeOffset > 0;
   462→
   463→  const goPrev = useCallback(() => {
   464→    if (canGoPrev) setWindowOffset((o) => o + 1);
   465→  }, [canGoPrev]);
   466→
   467→  const goNext = useCallback(() => {
   468→    if (canGoNext) setWindowOffset((o) => o - 1);
   469→  }, [canGoNext]);
   470→
   471→  // Compute day-type averages for the active window
   472→  const ktmbResult = useMemo(
   473→    () =>
   474→      computeDayTypeAverages<KtmbRow>(
   475→        ktmbData,
   476→        KTMB_SERVICES.map((s) => s.key),
   477→        activeWindow?.start,
   478→        activeWindow?.end
   479→      ),
   480→    [ktmbData, activeWindow]
   481→  );
   482→
   483→  const prasResult = useMemo(
   484→    () =>
   485→      computeDayTypeAverages<PrasaranaRow>(
   486→        prasData,
   487→        PRASARANA_LINES.map((s) => s.key),
   488→        activeWindow?.start,
   489→        activeWindow?.end
   490→      ),
   491→    [prasData, activeWindow]
   492→  );
   493→
   494→  // Current result based on active tab
   495→  const result = activeTab === 'ktmb' ? ktmbResult : prasResult;
   496→  const series = activeTab === 'ktmb' ? KTMB_SERVICES : PRASARANA_LINES;
   497→
   498→  // Compute previous window result for delta comparison
   499→  const prevWindow = windows[safeOffset + 1];
   500→  const prevResult = useMemo(() => {
   501→    if (!prevWindow) return null;
   502→    if (activeTab === 'ktmb') {
   503→      return computeDayTypeAverages<KtmbRow>(
   504→        ktmbData,
   505→        KTMB_SERVICES.map((s) => s.key),
   506→        prevWindow.start,
   507→        prevWindow.end
   508→      );
   509→    }
   510→    return computeDayTypeAverages<PrasaranaRow>(
   511→      prasData,
   512→      PRASARANA_LINES.map((s) => s.key),
   513→      prevWindow.start,
   514→      prevWindow.end
   515→    );
   516→  }, [prevWindow, activeTab, ktmbData, prasData]);
   517→
   518→  // Delta vs previous window
   519→  const weekdayAvgDelta = useMemo(() => {
   520→    if (!prevResult || prevResult.weekdayAvg === 0) return null;
   521→    return ((result.weekdayAvg - prevResult.weekdayAvg) / prevResult.weekdayAvg) * 100;
   522→  }, [result.weekdayAvg, prevResult]);
   523→
   524→  // Toggle line visibility
   525→  const toggleLine = useCallback((key: string) => {
   526→    setHiddenLines((prev) => {
   527→      const next = new Set(prev);
   528→      if (next.has(key)) next.delete(key);
   529→      else next.add(key);
   530→      return next;
   531→    });
   532→  }, []);
   533→
   534→  const showAll = useCallback(() => setHiddenLines(new Set()), []);
   535→  const hideAll = useCallback(
   536→    () => setHiddenLines(new Set(series.map((s) => s.key))),
   537→    [series]
   538→  );
   539→
   540→  // Format helper for Y-axis
   541→  const formatY = (v: number) => {
   542→    if (v >= 1000000) return `${(v / 1000000).toFixed(1)}M`;
   543→    if (v >= 1000) return `${(v / 1000).toFixed(0)}k`;
   544→    return v.toString();
   545→  };
   546→
   547→  // Dynamic lag label
   548→  const lagLabel = useMemo(() => {
   549→    if (!meta?.freshest_date) return '';
   550→    const dateMs = new Date(meta.freshest_date + 'T00:00:00').getTime();
   551→    const hoursAgo = Math.round((Date.now() - dateMs) / 36e5);
   552→    if (hoursAgo < 24) return `${hoursAgo}h ago`;
   553→    return `${Math.round(hoursAgo / 24)}d ago`;
   554→  }, [meta]);
   555→
   556→  if (loading) return <ChartSkeleton />;
   557→
   558→  if (!result.chartData.length) {
   559→    return (
   560→      <div className="rounded-2xl border border-[var(--border-subtle)] bg-[var(--surface-card)] backdrop-blur-md flex items-center justify-center h-[520px]">
   561→        <div className="text-center">
   562→          <BarChart3 className="w-8 h-8 text-[var(--text-ghost)] mx-auto mb-2" />
   563→          <p className="text-[var(--text-faint)] text-sm">
   564→            No daily data available for day-type analysis
   565→          </p>
   566→        </div>
   567→      </div>
   568→    );
   569→  }
   570→
   571→  const deltaIsPositive = weekdayAvgDelta !== null && weekdayAvgDelta > 0;
   572→  const deltaIsNegative = weekdayAvgDelta !== null && weekdayAvgDelta < 0;
   573→
   574→  return (
   575→    <div
   576→      data-chart
